mergeInto(LibraryManager.library, {
    OpenLinkInNewTab: function (urlPtr) {
        var url = UTF8ToString(urlPtr).trim();

        // Ensure it's absolute URL
        if (!/^https?:\/\//i.test(url)) {
            url = "https://" + url;
        }

        window.open(url, "_blank"); // open in new tab
    }
});
